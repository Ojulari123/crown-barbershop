def scan():
    rows = []
    for tree, root in TREES.items():
        for dp, _, fs in os.walk(root):
            for f in sorted(fs):
                if not f.endswith(('.ts', '.tsx')):
                    continue
                path = os.path.join(dp, f)
                for i, line in enumerate(open(path, encoding='utf-8'), 1):
                    rows.append((tree, f, i, line.rstrip('\n'), path))
    return rows


def main():
    rows = scan()
    out = []
    counts = Counter()
    unmapped = []

    def call_pat(k):
        return re.compile(r'\b' + k + r'\(') if k[0].islower() else re.compile(r'\b' + k + r'\b')

    # stateful API
    out.append('## A. Stateful store API (every call outside store.ts)\n')
    out.append('| # | Where | Call | Reads / writes | Endpoint |')
    out.append('|---|---|---|---|---|')
    n = 0
    for tree, f, i, line, _ in rows:
        if f == 'store.ts' or line.lstrip().startswith('import'):
            continue
        for k in STATEFUL:
            hits = len(call_pat(k).findall(line))
            if not hits:
                continue
            counts[(tree, k)] += hits
            key = (tree, f, i)
            if k == 'useCrown':
                m = USECROWN.get(key)
            elif k == 'setSlice':
                m = SETSLICE.get(key)
            else:
                m = OTHER.get(key)
            if m is None:
                unmapped.append((tree, f, i, k))
                m = ('UNMAPPED', 'UNMAPPED')
            n += 1
            out.append(f'| {n} | `{tree}/{f}:{i}` | `{k}` | {m[0]} | {m[1]} |')
    out.append('')

    out.append('## B. Pure store helpers (every use outside store.ts, imports excluded)\n')
    out.append('| # | Where | Helper | Endpoint |')
    out.append('|---|---|---|---|')
    n = 0
    for tree, f, i, line, _ in rows:
        if f == 'store.ts' or line.lstrip().startswith('import'):
            continue
        for k in PURE:
            hits = len(call_pat(k).findall(line))
            if not hits:
                continue
            counts[(tree, k)] += hits
            n += 1
            mult = f' (x{hits})' if hits > 1 else ''
            out.append(f'| {n} | `{tree}/{f}:{i}` | `{k}`{mult} | {PURE_MAP[k]} |')
    out.append('')

    out.append('## C. Code that bypasses the store or hard-codes demo behaviour\n')
    out.append('| # | Where | Pattern | Action in the port |')
    out.append('|---|---|---|---|')
    n = 0
    for tree, f, i, line, _ in rows:
        if f == 'store.ts':
            continue
        for p in BYPASS:
            if re.search(p, line):
                n += 1
                label = p.replace('\\b', '').replace('\\', '')
                out.append(f'| {n} | `{tree}/{f}:{i}` | `{label}` | {BYPASS_MAP[p]} |')
    out.append('')

    out.append('## D. Import lines from ./store (for the port checklist)\n')
    out.append('| Where | Imported names |')
    out.append('|---|---|')
    for tree, f, i, line, _ in rows:
        if "from './store'" in line:
            names = re.search(r'import (?:type )?\{(.*)\}', line)
            out.append(f'| `{tree}/{f}:{i}` | {names.group(1).strip() if names else line.strip()} |')
    out.append('')

    out.append('## E. Occurrence counts produced by this scan\n')
    out.append('| Identifier | site | admin |')
    out.append('|---|---|---|')
    for k in STATEFUL + PURE:
        out.append(f"| `{k}` | {counts[('site', k)]} | {counts[('admin', k)]} |")
    out.append('')
    if unmapped:
        out.append('**UNMAPPED rows:** ' + ', '.join(f'{t}/{f}:{i} {k}' for t, f, i, k in unmapped))
    sys.stdout.write('\n'.join(out) + '\n')
    sys.stderr.write(f'unmapped={len(unmapped)}\n')


if __name__ == '__main__':
    main()
